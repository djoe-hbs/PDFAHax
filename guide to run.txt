# 1. Extract  (PyMuPDF -> structured blocks; splits list items into list_item)
python extract_pdf.py Final_Test_Input.pdf

# 2. Detect tables  (Docling TableFormer -> table regions)
python detect_tables.py Final_Test_Input.pdf --output table_regions.json

# 3. Merge  (replace in-table blocks with per-cell table_cell blocks)
python merge_tables.py "output/Final_Test_Input/Final_Test_Input_structured_blocks.json" table_regions.json

# 3a. Footnote candidate detection (CODE finds candidates by geometry; the AI
#     CONFIRMS/REJECTS each one in step 4 using text context — see prompt rule
#     9a). Writes page_N_footnote_candidates.json per page (empty list = none
#     on that page, which is normal and requires no AI action) PLUS
#     footnote_geometry.json (code-facing: glyph bboxes for step 5b's precise
#     /Rect placement — the AI never sees this file, only the lean per-page one).
#     This is a REVIEW step too: read the printed table before sending
#     blocks_for_ai.json to the AI, so you know what candidates it will see.
python detect_footnotes.py Final_Test_Input.pdf --blocks-json "output/Final_Test_Input/Final_Test_Input_structured_blocks_merged.json" --output footnote_pairing_map.json --candidates-dir footnote_candidates --geometry-output footnote_geometry.json

# 4. Build the slim AI input, then send blocks_for_ai.json + the `prompt` file
#    + the relevant page_N_footnote_candidates.json file(s) to the AI. Save
#    its returned JSON as tags_merged.json
#    (the AI output may be a flat list OR the {tagged_blocks:[...]} format —
#     inject/visualize accept both.)
python -c "
import json
d=json.load(open('output/Final_Test_Input/Final_Test_Input_structured_blocks_merged.json',encoding='utf-8'))
slim=[]
for b in d['blocks']:
    e={'block_id':b['block_id'],'type':b['type'],'text':b['text'][:120],'bbox':b['bbox'],'metadata':b.get('metadata',{})}
    if b['type']=='table_cell':
        e['is_header']=b['is_header']; e['row']=b['row']; e['col']=b['col']
    slim.append(e)
json.dump(slim,open('blocks_for_ai.json','w'),indent=2)
print(len(slim),'blocks')
"

# 5. Inject  (real PDF/UA tags into the PDF binary)
#    - Tags matched text + wraps leftover content as Artifact (nothing left bare)
#    - Per-page MCIDs + multi-run marked content (ParentTree resolves correctly)
#    - Tables get ColSpan/RowSpan + TH Scope; sets PDF/UA id + DisplayDocTitle
#    - Repairs pre-existing malformed source /Artifact marks that straddle
#      BT/ET/q/Q (ISO 32000-2 Figure 9 violations some authoring tools write)
python inject_tags.py Final_Test_Input.pdf "output/Final_Test_Input/Final_Test_Input_structured_blocks_merged.json" tags_merged.json --output tagged_output.pdf

# 5a. TOC clickable links (OPTIONAL, two-step approve-then-build flow):
#     Step A — build the destination map and REVIEW it before anything touches
#     the PDF. Matches each TOCI's text to its body heading (H1-H6) by TEXT, not
#     printed page number (which is only a sanity-check column in the table).
python build_toc_destination_map.py "output/Final_Test_Input/Final_Test_Input_structured_blocks_merged.json" tags_merged.json --output toc_destination_map.json
#     -> prints a review table: TOC entry | matched heading | body page | printed
#        page | confidence (HIGH/MEDIUM/LOW/AMBIGUOUS/UNMATCHED). READ IT. An
#        UNMATCHED entry is safe (stays a plain TOCI, no guessed link) — an
#        AMBIGUOUS or LOW one should be checked by hand before proceeding.
#
#     Step B — ONLY after you've reviewed and approved toc_destination_map.json,
#     re-run inject_tags.py with --toc-map to create the GoTo link annotations
#     (wired TOCI -> Link -> MCR+OBJR, same mechanism as any other page link):
python inject_tags.py Final_Test_Input.pdf "output/Final_Test_Input/Final_Test_Input_structured_blocks_merged.json" tags_merged.json --output tagged_output.pdf --toc-map toc_destination_map.json

# 5b. Footnote reference links (approve-then-build, same discipline as TOC
#     links — the map/candidates were already reviewed in step 3a; this step
#     builds ONLY for pairs the AI actually confirmed in tags_merged.json):
#     - Reads footnote_geometry.json (code's geometric candidates) and
#       cross-checks each against tags_merged.json: a Reference/Note pair only
#       gets wired if BOTH sides are AI-confirmed (tag + matching
#       pairs_with_block_id on both). Rejected/unmatched candidates are
#       silently skipped — no guessed link, the block stays plain text.
#     - Builds Reference -> Link(marker MCID + OBJR) + Note -> (Lbl + body),
#       reusing the TOC-link annotation mechanism. The marker's ops are split
#       from the surrounding paragraph BY FONT SIZE (the superscript is its
#       own content-stream operation, ~65% the size of body text) so the
#       Link's MCID wraps ONLY the tiny glyph, never adjacent normal text.
#     - REUSES a pre-existing link annotation if the source PDF already has
#       one covering the marker glyph (common — many authoring tools already
#       create a footnote GoTo link) instead of creating a duplicate.
python inject_tags.py Final_Test_Input.pdf "output/Final_Test_Input/Final_Test_Input_structured_blocks_merged.json" tags_merged.json --output tagged_output.pdf --toc-map toc_destination_map.json --footnote-geometry footnote_geometry.json

# 6. Visualize  (draw colored tag boxes for a quick eyeball check)
python visualize_tags.py Final_Test_Input.pdf "output/Final_Test_Input/Final_Test_Input_structured_blocks_merged.json" tags_merged.json --output tagged_visual.pdf

# 7. Validate  (veraPDF PDF/UA-1 gate — prints pass/fail per rule, worst first)
#    Saves tagged_output_verapdf.xml next to the PDF. Exit code 0 = compliant.
python validate_pdf.py tagged_output.pdf
